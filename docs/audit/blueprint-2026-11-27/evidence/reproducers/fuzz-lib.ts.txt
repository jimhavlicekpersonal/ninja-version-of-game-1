export function mulberry32(a: number) {
  return function () {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
export type Rng = () => number;
export const pick = <T>(r: Rng, xs: readonly T[]): T => xs[Math.floor(r() * xs.length)];
export const int = (r: Rng, lo: number, hi: number) => lo + Math.floor(r() * (hi - lo + 1));

export const PROTO_KEYS = ['__proto__', 'constructor', 'toString', 'hasOwnProperty', 'valueOf', 'prototype', 'length'];
export const BUILDING_IDS = Object.keys(data.buildings);
export const TROOP_IDS = Object.keys(data.troops);
export const HERO_IDS = Object.keys(data.battle.heroes ?? {});
export const PLANNED_BUILDINGS = Object.keys((planned as { buildings: object }).buildings);
export const PLANNED_HEROES = Object.keys((planned as { heroes: object }).heroes);
export const ABILITY_IDS = Object.keys((data.battle as unknown as { abilities?: object }).abilities ?? {});

const WEIRD_NUMS = [NaN, Infinity, -Infinity, -0, 0, 1, -1, 0.5, 1.5, -0.5, 1e308, -1e308, 2 ** 53 + 1, -(2 ** 53), 1e21, 2 ** 31, 2 ** 32, 4294967295, 1e-9, 5e-324, 99, 1000, 100000, 2];
const WEIRD_OTHER: unknown[] = [undefined, null, '', 'x', '1', '0', ' ', 'b1', [], [1], {}, { a: 1 }, true, false, '__proto__', 'constructor', 'toString', 'hasOwnProperty', ['b1'], { toString: 1 }, { valueOf: 1 }, '1e999', 'NaN'];
/** A JSON-reachable subset (what a server could receive). */
const WEIRD_JSON_NUMS = [Infinity, -Infinity, -0, 0, 1, -1, 0.5, 1.5, -0.5, 1e308, -1e308, 2 ** 53 + 1, -(2 ** 53), 1e21, 2 ** 31, 2 ** 32, 4294967295, 1e-9, 5e-324, 99, 1000, 100000, 2];
const WEIRD_JSON_OTHER: unknown[] = [null, '', 'x', '1', '0', ' ', 'b1', [], [1], {}, { a: 1 }, true, false, '__proto__', 'constructor', 'toString', 'hasOwnProperty', ['b1'], { toString: 1 }, { valueOf: 1 }];

export function weird(r: Rng, jsonOnly = false): unknown {
  if (jsonOnly) return r() < 0.55 ? pick(r, WEIRD_JSON_NUMS) : pick(r, WEIRD_JSON_OTHER);
  return r() < 0.55 ? pick(r, WEIRD_NUMS) : pick(r, WEIRD_OTHER);
}
export function weirdNum(r: Rng, jsonOnly = false): number {
  return jsonOnly ? pick(r, WEIRD_JSON_NUMS) : pick(r, WEIRD_NUMS);
}

export function anyBuildingType(r: Rng): string {
  const x = r();
  if (x < 0.7) return pick(r, BUILDING_IDS);
  if (x < 0.8) return pick(r, PLANNED_BUILDINGS);
  if (x < 0.87) return pick(r, TROOP_IDS);
  if (x < 0.95) return pick(r, PROTO_KEYS);
  return 'no_such_building';
}
export function anyTroop(r: Rng): string {
  const x = r();
  if (x < 0.72) return pick(r, TROOP_IDS);
  if (x < 0.77) return pick(r, HERO_IDS.length ? HERO_IDS : TROOP_IDS);
  if (x < 0.81) return pick(r, PLANNED_HEROES);
  if (x < 0.85) return pick(r, BUILDING_IDS);
  if (x < 0.88 && ABILITY_IDS.length) return pick(r, ABILITY_IDS);
  if (x < 0.96) return pick(r, PROTO_KEYS);
  return 'no_such_troop';
}
export function anyBuildingId(r: Rng, s: VillageState): string {
  const x = r();
  if (x < 0.85 && s.buildings.length) return pick(r, s.buildings).id;
  if (x < 0.9) return `b${int(r, 0, s.nextId + 3)}`;
  if (x < 0.97) return pick(r, PROTO_KEYS);
  return 'nope';
}

export function freeSpot(r: Rng, s: VillageState, type: string): { x: number; y: number } {
  const map = data.economy.mapSize;
  if (data.buildings[type] && Object.hasOwn(data.buildings, type)) {
    for (let i = 0; i < 40; i++) {
      const x = int(r, 0, map - 1);
      const y = int(r, 0, map - 1);
      try {
        if (canPlace(s, data, type, x, y).ok) return { x, y };
      } catch {
        break;
      }
    }
  }
  return { x: int(r, -2, map + 1), y: int(r, -2, map + 1) };
}

export function randArmy(r: Rng, jsonOnly: boolean): Record<string, number> {
  const out: Record<string, number> = {};
  const n = int(r, 0, 5);
  for (let i = 0; i < n; i++) {
    const k = anyTroop(r);
    const v = r() < 0.75 ? int(r, 0, 30) : (weird(r, jsonOnly) as number);
    if (k === '__proto__') Object.defineProperty(out, k, { value: v, enumerable: true, writable: true, configurable: true });
    else out[k] = v;
  }
  return out;
}

/** Mutate one random field of a command to something weird. */
export function corrupt(r: Rng, cmd: Record<string, unknown>, jsonOnly: boolean): Record<string, unknown> {
  const keys = Object.keys(cmd).filter((k) => k !== 'type');
  const out: Record<string, unknown> = { ...cmd };
  const x = r();
  if (!keys.length || x < 0.1) {
    out[pick(r, ['extra', 'id', 'count', 'troop', 'x', 'pay', 'index', 'known', 'verified', 'economy'])] = weird(r, jsonOnly);
    return out;
  }
  const k = pick(r, keys);
  if (x < 0.25) delete out[k];
  else out[k] = weird(r, jsonOnly);
  return out;
}

export interface Problem {
  kind: string;
  detail: string;
}

const INT_PATHS = /^(resources\.|army\.|nextId$|nextSeed$|trophies$|season$|stats\.|trials\.best$|pass\.points$|pass\.season$|tasks\.day$|tasks\.week$|daily\.step$|daily\.lastDay$|stipend\.|calendar\.|buildings\.\d+\.(x|y|level|upgrade\.toLevel)$|research\.levels\.|grove\.reinforcements\.|grove\.reinforcementLevels\.)/;
const FRACTION_OK = /^(stats\.raidDestruction|buildings\.\d+\.collector\.stored)/;

function walk(v: unknown, path: string, out: Problem[], depth = 0): void {
  if (depth > 40) return;
  if (typeof v === 'number') {
    if (!Number.isFinite(v)) out.push({ kind: 'non-finite', detail: `${path}=${v}` });
    else if (Object.is(v, -0)) out.push({ kind: 'neg-zero', detail: path });
    else if (!Number.isInteger(v) && INT_PATHS.test(path) && !FRACTION_OK.test(path)) out.push({ kind: 'fractional', detail: `${path}=${v}` });
    else if (Math.abs(v) > Number.MAX_SAFE_INTEGER && INT_PATHS.test(path)) out.push({ kind: 'unsafe-int', detail: `${path}=${v}` });
    return;
  }
  if (v === undefined) {
    out.push({ kind: 'undefined-prop', detail: path });
    return;
  }
  if (v === null || typeof v !== 'object') {
    if (typeof v === 'function' || typeof v === 'symbol' || typeof v === 'bigint') out.push({ kind: 'non-json', detail: `${path}:${typeof v}` });
    return;
  }
  if (path.startsWith('battleLog') || path.startsWith('defenseLog')) {
    // logs: only shallow
    if (depth > 3) return;
  }
  if (Array.isArray(v)) {
    for (let i = 0; i < v.length; i++) walk(v[i], path ? `${path}.${i}` : String(i), out, depth + 1);
    return;
  }
  for (const k of Object.keys(v as object)) walk((v as Record<string, unknown>)[k], path ? `${path}.${k}` : k, out, depth + 1);
}

/** Commands after which Jade may legitimately rise (rewards, refunds of Jade top-ups, purchases). */
const JADE_GRANTERS = new Set<string>([
  'openDailyChest', 'claimTask', 'claimTier', 'claimAchievement', 'claimStipend', 'claimCalendar', 'trialEnd', 'bountyEnd', 'rematchEnd', 'drillEnd', 'claimKeepSeal',
  'cancel', 'cancelTraining', 'cancelResearch', 'grantPurchase', 'reinstatePurchase', 'staffAdjust', 'battleEnd', 'groveBattleEnd', 'claimGangChest', 'claimRaidChest', 'settleOnlineAttack', 'dismiss',
]);

export interface StepInfo {
  problems: Problem[];
  threw?: string;
  ok?: boolean;
  code?: string;
}

const strip = (s: VillageState) => {
  const c = structuredClone(s) as unknown as Record<string, unknown>;
  delete c.lastSeenAt;
  return c;
};

export function checkState(s: VillageState, now: number, opts: { negOk?: boolean } = {}): Problem[] {
  const out: Problem[] = [];
  walk(s, '', out);
  // validateVillage: would this village load again?
  let vp: string[] = [];
  try {
    vp = validateVillage(s, data);
  } catch (e) {
    out.push({ kind: 'validate-throws', detail: String((e as Error).message).slice(0, 200) });
  }
  if (vp.length) out.push({ kind: 'validate-fails', detail: vp.slice(0, 3).join('; ') });
  // resources
  let cap: Record<string, number> | null = null;
  try {
    cap = capacity(s, data) as unknown as Record<string, number>;
  } catch (e) {
    out.push({ kind: 'capacity-throws', detail: String((e as Error).message) });
  }
  for (const k of ['acorns', 'dew', 'nightsap', 'moonstones'] as const) {
    const v = s.resources[k];
    if (typeof v !== 'number') {
      out.push({ kind: 'resource-type', detail: `${k}:${typeof v}` });
      continue;
    }
    if (v < 0 && !opts.negOk) out.push({ kind: 'negative-resource', detail: `${k}=${v}` });
    if (cap && k !== 'moonstones' && v > cap[k] + 1e-6) out.push({ kind: 'over-cap', detail: `${k}=${v} cap=${cap[k]}` });
  }
  // army / housing
  try {
    const hc = housingCapacity(s, data);
    const ah = armyHousing(s.army, data);
    if (ah > hc) out.push({ kind: 'army-over-housing', detail: `${ah}>${hc}` });
    const qh = queueHousing(s.training, data);
    const lim = trainingQueueLimit(s, data);
    if (ah + qh > lim) out.push({ kind: 'queue-over-limit', detail: `${ah}+${qh}>${lim}` });
    for (const [t, n] of Object.entries(s.army)) if (!Number.isInteger(n) || n < 0) out.push({ kind: 'army-count', detail: `${t}=${n}` });
    // queue sanity: ordering
    let prevEnd = -Infinity;
    for (const q of s.training) {
      if (!(q.endsAt >= q.startedAt)) out.push({ kind: 'queue-timer', detail: `${q.troop} ${q.startedAt}->${q.endsAt}` });
      if (q.endsAt < prevEnd) out.push({ kind: 'queue-order', detail: `${q.troop} ends ${q.endsAt} before previous ${prevEnd}` });
      prevEnd = q.endsAt;
    }
  } catch (e) {
    out.push({ kind: 'army-throws', detail: String((e as Error).message) });
  }
  // builders
  try {
    const bs = builderStatus(s, data, now);
    if (bs.busy > bs.total) out.push({ kind: 'builders-over', detail: `${bs.busy}>${bs.total}` });
  } catch (e) {
    out.push({ kind: 'builder-throws', detail: String((e as Error).message) });
  }
  // research lanes
  try {
    const seen = new Set<string>();
    for (const a of s.research.active) {
      if (seen.has(a.troop)) out.push({ kind: 'research-dup', detail: a.troop });
      seen.add(a.troop);
    }
  } catch (e) {
    out.push({ kind: 'research-throws', detail: String((e as Error).message) });
  }
  // save round trip
  try {
    const file = serialize(s, now);
    const json = JSON.stringify(file);
    const back = deserialize(JSON.parse(json), data);
    if (!isDeepStrictEqual(strip(back), strip(s))) {
      out.push({ kind: 'roundtrip-differs', detail: firstDiff(strip(back), strip(s)) });
    }
    const h1 = economyHash(s);
    const h2 = economyHash(back);
    if (h1 !== h2) out.push({ kind: 'hash-unstable', detail: `${h1} vs ${h2}` });
    const again = deserialize(JSON.parse(JSON.stringify(serialize(back, now))), data);
    if (!isDeepStrictEqual(again, back)) out.push({ kind: 'roundtrip-not-idempotent', detail: firstDiff(again, back) });
  } catch (e) {
    out.push({ kind: 'roundtrip-throws', detail: String((e as Error).message).slice(0, 200) });
  }
  return out;
}

export function firstDiff(a: unknown, b: unknown, path = ''): string {
  if (isDeepStrictEqual(a, b)) return '';
  if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object') return `${path}: ${fmt(a)} vs ${fmt(b)}`;
  const ka = Object.keys(a as object);
  const kb = Object.keys(b as object);
  for (const k of new Set([...ka, ...kb])) {
    const va = (a as Record<string, unknown>)[k];
    const vb = (b as Record<string, unknown>)[k];
    if (!isDeepStrictEqual(va, vb)) return firstDiff(va, vb, path ? `${path}.${k}` : k);
  }
  return `${path}: prototypes/keys differ (${ka.length} vs ${kb.length} keys)`;
}
const fmt = (v: unknown) => {
  try {
    return Object.is(v, -0) ? '-0' : v === undefined ? 'undefined' : typeof v === 'number' ? String(v) : JSON.stringify(v)?.slice(0, 80);
  } catch {
    return '?';
  }
};

/** One fuzz step: apply, check everything. `prev` must stay untouched. */
export function step(prev: VillageState, cmd: Command, now: number, opts: { negOk?: boolean; deep?: boolean } = {}): { next: VillageState; info: StepInfo } {
  const info: StepInfo = { problems: [] };
  const before = JSON.stringify(prev);
  let r: ReturnType<typeof applyCommand>;
  try {
    r = applyCommand(prev, cmd, now, data);
  } catch (e) {
    info.threw = `${(e as Error)?.constructor?.name}: ${String((e as Error)?.message).slice(0, 160)}`;
    if (JSON.stringify(prev) !== before) info.problems.push({ kind: 'input-mutated-on-throw', detail: '' });
    return { next: prev, info };
  }
  if (JSON.stringify(prev) !== before) info.problems.push({ kind: 'input-mutated', detail: '' });
  info.ok = r.ok;
  if (!r.ok) {
    info.code = r.code;
    return { next: prev, info };
  }
  const s = r.state;
  // Jade never rises outside a grant.
  let settled: VillageState | null = null;
  try {
    settled = advance(prev, now, data).state;
  } catch {
    settled = null;
  }
  if (settled) {
    const type = (cmd as { type: string }).type;
    const dj = s.resources.moonstones - settled.resources.moonstones;
    if (dj > 0 && !JADE_GRANTERS.has(type)) info.problems.push({ kind: 'jade-rose', detail: `${type} +${dj}` });
    if (type === 'topUp' || type === 'skip' || type === 'skipTraining' || type === 'skipResearch' || type === 'buySkin') {
      if (dj > 0) info.problems.push({ kind: 'jade-rose-on-spend', detail: `${type} +${dj}` });
    }
  }
  if (opts.deep !== false) info.problems.push(...checkState(s, now, opts));
  return { next: s, info };
}
