out.friend_BanswersAtoC = s(await call('POST', `/v1/friends/requests/${rid}`, { token: B.token, body: { accept: true } }));
out.friend_BcancelsAtoC = s(await call('DELETE', `/v1/friends/requests/${rid}`, { token: B.token }));
out.friend_requestStillOpenForA = (await call('GET', '/v1/friends', { token: A.token })).json.outgoing.some((x) => x.id === rid);

out.friend_CacceptsA = s(await call('POST', `/v1/friends/requests/${rid}`, { token: C.token, body: { accept: true } }));

const inv = await call('POST', '/v1/squad/invites', { token: A.token, body: { friendId: C.id } });

out.gang_BacceptsCsInvite = s(await call('POST', `/v1/squad/invites/${invId}`, { token: B.token, body: { accept: true } }));
out.gang_AinvitesNonFriendB = s(await call('POST', '/v1/squad/invites', { token: A.token, body: { friendId: B.id } }));
out.gang_BtakesLeadership = s(await call('POST', '/v1/squad/leader', { token: B.token, body: { playerId: B.id } }));
await call('POST', `/v1/squad/invites/${invId}`, { token: C.token, body: { accept: true } });

out.gang_BreadsChat = s(await call('GET', '/v1/squad/chat', { token: B.token }));
out.duel_BticketOnStranger = s(await call('POST', '/v1/duels/ticket', { token: B.token, body: { friendId: A.id } }));
out.duel_BchallengesStranger = s(await call('POST', '/v1/duels/challenges', { token: B.token, body: { friendId: A.id } }));
// alliances: two alliances, a troop request and a message in X; B is in Y
const X = await call('POST', '/v1/groves', { token: A.token, body: { name: 'Idor Xray' } });
await call('POST', '/v1/groves', { token: B.token, body: { name: 'Idor Yankee' } });

out.alliance_BreadsOwnChatOnly = !(await call('GET', '/v1/grove/chat', { token: B.token })).json.messages.some((m) => m.id === msg.json.id);
